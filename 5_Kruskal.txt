class DisjointSet:
    def __init__(self, vertices):
        self.parent = {v: v for v in vertices}
        self.rank = {v: 0 for v in vertices}

    def find(self, u):
        if self.parent[u] != u:
            self.parent[u] = self.find(self.parent[u])
        return self.parent[u]

    def union(self, u, v):
        root_u = self.find(u)
        root_v = self.find(v)
        if root_u == root_v:
            return False
        if self.rank[root_u] < self.rank[root_v]:
            self.parent[root_u] = root_v
        elif self.rank[root_u] > self.rank[root_v]:
            self.parent[root_v] = root_u
        else:
            self.parent[root_v] = root_u
            self.rank[root_u] += 1
        return True

def kruskal_mst(vertices, edges):
    ds = DisjointSet(vertices)
    mst = []
    edges.sort(key=lambda x: x[2])  # Sort by weight

    for u, v, w in edges:
        if ds.union(u, v):
            mst.append((u, v, w))
            if len(mst) == len(vertices) - 1:
                break

    return mst

# ✅ Input: Adjacency List
adjacency_list = {
    0: [(1, 4), (7, 8)],
    1: [(0, 4), (2, 8), (7, 11)],
    2: [(1, 8), (3, 7), (5, 4), (8, 2)],
    3: [(2, 7), (4, 9), (5, 14)],
    4: [(3, 9), (5, 10)],
    5: [(2, 4), (3, 14), (4, 10), (6, 2)],
    6: [(5, 2), (7, 1), (8, 6)],
    7: [(0, 8), (1, 11), (6, 1), (8, 7)],
    8: [(2, 2), (6, 6), (7, 7)],
}

# 🔄 Convert to Edge List
edges = []
added = set()
for u in adjacency_list:
    for v, w in adjacency_list[u]:
        if (v, u) not in added:
            edges.append((u, v, w))
            added.add((u, v))

# 🔢 Vertices
vertices = list(adjacency_list.keys())

# 🌳 Kruskal's Algorithm
mst = kruskal_mst(vertices, edges)

# 🖨️ Print MST
print("Minimum Spanning Tree:")
print("-----------------------")
total_weight = 0
for u, v, w in mst:
    print(f"{u} -- {v}  (weight = {w})")
    total_weight += w
print("-----------------------")
print(f"Total weight of MST: {total_weight}")
